      ```ts
      import { onCall, HttpsError } from 'firebase-functions/v2/https'
      import { CloudTasksClient } from '@google-cloud/tasks'

      interface ReleaseCallablePayload { reservationId: string; giverEmail?: string }
      const tasksClient = new CloudTasksClient()

      export const releaseReservationCallable = onCall<ReleaseCallablePayload>(
        { region: REGION },
        async (request) => {
          const reservationId = request.data?.reservationId
          if (!reservationId) throw new HttpsError('invalid-argument', 'MISSING_RESERVATION_ID')

          const db = admin.firestore()
          const ref = db.collection('reservations').doc(reservationId)
          let cloudTaskName = ''

          // Ownership check in a read-only transaction so we have a consistent view of status + giverId/giverEmail.
          await db.runTransaction(async (tx) => {
            const snap = await tx.get(ref)
            if (!snap.exists) throw new HttpsError('not-found', 'RESERVATION_NOT_FOUND')
            const data = snap.data()!
            if (data.status !== 'active') throw new HttpsError('failed-precondition', 'RESERVATION_NOT_ACTIVE')

            // Ownership: signed-in → uid must match giverId. Guest → giverId must be null AND payload.giverEmail must match.
            if (request.auth) {
              if (request.auth.uid !== data.giverId) {
                throw new HttpsError('permission-denied', 'RELEASE_NOT_OWNER')
              }
            } else {
              const payloadEmail = request.data?.giverEmail
              if (!payloadEmail || data.giverId !== null || data.giverEmail !== payloadEmail) {
                throw new HttpsError('permission-denied', 'RELEASE_NOT_OWNER')
              }
            }
            cloudTaskName = (data.cloudTaskName as string) ?? ''
          })

          // Now do the real release with the manual-release flag. releaseReservationCore is itself transactional
          // and idempotent — running back-to-back is safe.
          await releaseReservationCore({
            reservationId,
            db: admin.firestore(),
            skipNotYetExpiredGuard: true,
          })

          // Cancel the scheduled Cloud Task AFTER release (mirror confirmPurchase pattern — swallow NOT_FOUND).
          if (cloudTaskName) {
            try { await tasksClient.deleteTask({ name: cloudTaskName }) }
            catch (err: unknown) {
              const code = (err as { code?: number }).code
              if (code !== 5) console.error('[releaseReservationCallable] deleteTask failed', err)
            }
          }
          return { success: true }
        }
      )
      ```

    STEP B — Create `web/src/features/reservation/useActiveReservationHydration.ts`.
      Effect-driven hook (no return value beyond a `status: 'idle'|'loading'|'hydrated'|'empty'|'error'`). Calls the `hydrateActiveReservation` callable when:
        - `authReady === true` (gate on Firebase Auth — pitfall 7 from useAuth.ts comment)
        - `registryId` is non-empty
        - Either `user` (signed-in) OR `identity` (guest with email in localStorage) is present
        - `active === null` (DO NOT clobber a fresh in-session reservation — constraint)
      AND a guard ref prevents double-fire (StrictMode + dependency churn).

      On success:
        - If response.active is non-null → call `setActive(response.active)`.
        - Else → do nothing (leave active null).

      Cleanup-safe (constraint): when the effective key changes (registryId, user.uid, identity.email, user transitions in/out), reset the guard ref so a refetch fires next time. Use a `lastKeyRef` of the form `${registryId}|${user?.uid ?? 'guest'}|${user?.email ?? identity?.email ?? ''}` to detect.

      Skeleton:
      ```ts
      import { useEffect, useRef, useState } from 'react'
      import { httpsCallable } from 'firebase/functions'
      import { functions } from '../../firebase'
      import { useAuth } from '../auth/useAuth'
      import { useGuestIdentity } from '../auth/useGuestIdentity'
      import { useActiveReservation, type ActiveReservation } from './useActiveReservation'

      type HydrationStatus = 'idle' | 'loading' | 'hydrated' | 'empty' | 'error'

      interface HydrationResponse { active: ActiveReservation | null }

      export function useActiveReservationHydration(registryId: string | undefined) {
        const { user, isReady: authReady } = useAuth()
        const { identity } = useGuestIdentity()
        const { active, set } = useActiveReservation()
        const [status, setStatus] = useState<HydrationStatus>('idle')
        const lastKeyRef = useRef<string | null>(null)

        useEffect(() => {
          if (!registryId) return
          if (!authReady) return
          if (active) return                                    // don't clobber an in-session reservation
          const effectiveEmail = user?.email ?? identity?.email ?? null
          if (!user && !identity) return                        // truly anonymous and no stored guest identity → cannot hydrate
          if (!effectiveEmail) return

          const key = `${registryId}|${user?.uid ?? 'guest'}|${effectiveEmail}`
          if (lastKeyRef.current === key) return
          lastKeyRef.current = key

          setStatus('loading')
          const callable = httpsCallable<
            { registryId: string; giverEmail?: string },
            HydrationResponse
          >(functions, 'hydrateActiveReservation')

          // Signed-in path sends ONLY registryId — backend uses auth.uid (defence in depth).
          // Guest path sends registryId + giverEmail.
          const payload = user
            ? { registryId }
            : { registryId, giverEmail: effectiveEmail }

          let cancelled = false
          callable(payload).then((r) => {
            if (cancelled) return
            if (r.data?.active) {
              set(r.data.active)
              setStatus('hydrated')
            } else {
              setStatus('empty')
            }
          }).catch((err) => {
            if (cancelled) return
            // Hydration is best-effort: a failure here must not block page render.
            // Log only; banner stays hidden until the user reserves again.
            // eslint-disable-next-line no-console
            console.warn('[useActiveReservationHydration] failed', err)
            setStatus('error')
          })

          return () => { cancelled = true }
        }, [registryId, authReady, user, identity, active, set])

        return { status }
      }
      ```
